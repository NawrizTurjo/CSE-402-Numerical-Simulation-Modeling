\section{Eigenvalue Decomposition (Power Method)}

\begin{formulabox}
Defining/characteristic equation:
\[
A\mathbf v = \lambda\mathbf v \ \Rightarrow\ (A-\lambda I)\mathbf v = 0 \ \Rightarrow\ \det(A-\lambda I)=0
\]
Eigen decomposition ($V=[\mathbf v_1\cdots\mathbf v_n]$, $\Lambda=\mathrm{diag}(\lambda_i)$):
\[
AV = V\Lambda \ \Rightarrow\ A = V\Lambda V^{-1}, \qquad A^k = V\Lambda^k V^{-1}
\]
\textbf{Power method} ($\mathbf x=\sum c_i\mathbf v_i$, needs $|\lambda_1|>|\lambda_2|\ge\cdots$):
\[
A^k\mathbf x = \lambda_1^k\left[c_1\mathbf v_1 + \sum_{i\ge2} c_i\left(\frac{\lambda_i}{\lambda_1}\right)^k\mathbf v_i\right] \to \lambda_1^k c_1\mathbf v_1
\]
Iteration:
\[
\mathbf y_{k+1}=A\mathbf x_k, \qquad \mathbf x_{k+1} = \frac{\mathbf y_{k+1}}{\text{largest-magnitude entry}}
\]
\[
\lambda_1\approx\text{that largest-magnitude entry}
\]
\[
|\varepsilon_a| = \left|\frac{\lambda_k-\lambda_{k-1}}{\lambda_k}\right|\times100\%
\]
\textbf{Inverse power method} (smallest eigenvalue): $A\mathbf v=\lambda\mathbf v \Rightarrow A^{-1}\mathbf v = \tfrac1\lambda\mathbf v$; run power method on $A^{-1}$ (solve $A\mathbf y=\mathbf x$ via LU), $\lambda_{\min}\approx1/(\text{normalizing factor})$.

\textbf{Deflation} (intermediate eigenvalues, symmetric $A$): $\hat{\mathbf v}_1=\mathbf v_1/\sqrt{\mathbf v_1^\top\mathbf v_1}$,
\[
A_2 = A - \lambda_1\hat{\mathbf v}_1\hat{\mathbf v}_1^\top
\]
power method on $A_2\to\lambda_2$; repeat $A_3=A_2-\lambda_2\hat{\mathbf v}_2\hat{\mathbf v}_2^\top$, etc.
\end{formulabox}
